let audioCtx; let soundTimer; let soundFrame; let soundActive = false; function initAudio() { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } if (audioCtx.state === "suspended") { audioCtx.resume(); } } // Ein knackiges mechanisches Rad-Klicken function playClick(intensity = 1) { if (!audioCtx || !soundActive) return; const now = audioCtx.currentTime; // Tiefer, kurzer mechanischer Anschlag const body = audioCtx.createOscillator(); const bodyGain = audioCtx.createGain(); body.type = "triangle"; body.frequency.setValueAtTime(180, now); body.frequency.exponentialRampToValueAtTime(65, now + 0.045); bodyGain.gain.setValueAtTime(0.0001, now); bodyGain.gain.exponentialRampToValueAtTime(0.22 * intensity, now + 0.003); bodyGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.055); body.connect(bodyGain); bodyGain.connect(audioCtx.destination); body.start(now); body.stop(now + 0.06); // Heller Klick für den typischen Glücksrad-Sound const click = audioCtx.createOscillator(); const clickGain = audioCtx.createGain(); click.type = "square"; click.frequency.setValueAtTime(1250, now); click.frequency.exponentialRampToValueAtTime(420, now + 0.018); clickGain.gain.setValueAtTime(0.0001, now); clickGain.gain.exponentialRampToValueAtTime(0.12 * intensity, now + 0.002); clickGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.025); click.connect(clickGain); clickGain.connect(audioCtx.destination); click.start(now); click.stop(now + 0.03); } function startWheelSound(duration = 5000) { initAudio(); soundActive = true; const start = performance.now(); let lastClick = 0; function tick(now) { if (!soundActive) return; const progress = Math.min((now - start) / duration, 1); // Wird beim Abbremsen immer langsamer const interval = 55 + Math.pow(progress, 2.8) * 520; if (now - lastClick >= interval) { const intensity = 1 - progress * 0.25; playClick(intensity); lastClick = now; } if (progress < 1) { soundFrame = requestAnimationFrame(tick); } else { // Satter Abschluss-Klick setTimeout(() => { if (soundActive) playClick(1.5); }, 100); } } soundFrame = requestAnimationFrame(tick); } function stopWheelSound() { soundActive = false; if (soundFrame) cancelAnimationFrame(soundFrame); }
